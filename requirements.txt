# ─────────────────────────────────────────────────────────────────────────────
# 🎭 BROWSER AUTOMATION & TEST FRAMEWORK
# ─────────────────────────────────────────────────────────────────────────────

# Playwright — modern browser automation (Microsoft).
# Drives Chromium/Firefox/WebKit headlessly or with UI.
# Why this and not Selenium? Faster, auto-waits, better DX, built-in tracing.
playwright==1.48.0

# Pytest — the de-facto Python testing framework.
# Provides fixtures, assertions, parametrization, plugin ecosystem.
pytest==8.3.3

# pytest-bdd — BDD plugin for pytest.
# Lets us write Gherkin .feature files (Given/When/Then) and map each
# step to Python functions. Enables business-readable tests.
pytest-bdd==7.3.0

# pytest-playwright — official integration of Playwright with pytest.
# Adds the `page`, `browser`, `context` fixtures and CLI flags.
pytest-playwright==0.5.2

# Allure for pytest — generates rich HTML test reports.
# Captures screenshots, traces, durations, and trends across runs.
allure-pytest==2.13.5

# pytest-html — self-contained single-file HTML report.
# Used by Agent 4 (Runner) for human-readable artifact uploaded to GitHub Actions.
# Read by Agent 5 (Reporter) is via junit.xml, not this HTML.
pytest-html==4.1.1


# ─────────────────────────────────────────────────────────────────────────────
# 🤖 LLM & AGENT ORCHESTRATION
#
# IMPORTANT: All these packages must be on the same major version (1.x).
# In Dec 2025, langchain-google-genai 4.0.0 was released using the unified
# google-genai SDK. It needs langchain-core 1.x, which needs langchain 1.x.
# Upgrading any one of these requires upgrading all of them together.
# ─────────────────────────────────────────────────────────────────────────────

# LangChain — high-level abstractions over LLM APIs.
# We use it for: prompt templates, structured output, message types.
# Standardizes how all 5 agents talk to Gemini.
langchain>=1.0,<2.0

# langchain-core — the foundation that everything else builds on.
# Pinned to 1.x to match langchain-google-genai 4.x requirements.
langchain-core>=1.0,<2.0

# LangGraph — state machine library for orchestrating multi-agent workflows.
# Each agent is a node; edges define transitions; supports human-in-the-loop
# interrupts. We use this in Day 6 to wire all agents into one pipeline.
langgraph>=1.0,<2.0

# LangChain wrapper for Google Gemini.
# CRITICAL: must be >= 4.0.0 because earlier versions don't support the
# json_schema method for with_structured_output(), which is essential for
# reliable structured output with Gemini 2.5 models. Earlier versions
# silently return None when the model doesn't call the tool.
langchain-google-genai>=4.0.0,<5.0

# Pydantic — data validation library used everywhere in our agents.
# Defines the SHAPE of LLM outputs so we never trust raw text from the LLM.
# If Gemini returns malformed data, Pydantic raises a clear error.
pydantic>=2.9


# ─────────────────────────────────────────────────────────────────────────────
# 🔌 EXTERNAL SERVICE INTEGRATIONS
# ─────────────────────────────────────────────────────────────────────────────

# Jira Python SDK — official Atlassian-supported client.
# Used by JiraClient to read stories, create Test Case issues, manage labels.
# >=3.10.5 required because Atlassian removed the legacy /rest/api/2/search
# endpoint in 2025; older versions raise HTTP 410 errors.
jira>=3.10.5

# PyGithub — GitHub REST API client.
# Used by Agent 2 (Coder) on Day 4 to create branches, commit files,
# and open pull requests programmatically.
# Used by Agent 3 (Reviewer) on Day 5 to read PRs and post review comments.
PyGithub>=2.4

# Requests — HTTP client. Indirect dependency for many libraries above,
# but we list it explicitly because we may use it for ad-hoc HTTP calls.
requests>=2.32

# Tenacity — explicit dependency for our @retry decorators in llm_client.py.
# Auto-installed by langchain, but we list it because we use it directly.
tenacity>=8.0


# ─────────────────────────────────────────────────────────────────────────────
# ⚙️ CONFIGURATION & UTILITIES
# ─────────────────────────────────────────────────────────────────────────────

# python-dotenv — loads `.env` files into os.environ at runtime.
# Keeps secrets out of code; pairs with `.env.example` for documentation.
python-dotenv>=1.0

# PyYAML — read/write YAML files.
# config.yaml uses YAML for human-friendly config (browsers, test users,
# per-agent selection rules).
pyyaml>=6.0
